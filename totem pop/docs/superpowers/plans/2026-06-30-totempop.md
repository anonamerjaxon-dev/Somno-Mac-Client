**Goal:** Build a Swift Package `TotemPop` that monitors accelerometer input and key combinations, then plays a Minecraft totem of undying pop animation as a full-screen transparent overlay on top of everything.

**Architecture:** A Swift Package with a self-contained library (overlay, player, trigger monitoring, rate limiting, config) and a demo app target. The library owns its background thread, monitors the BMI286 accelerometer via IOKit, listens for global hotkeys via CGEventTap, and displays the totem via a maximum-window-level NSWindow with AVPlayer.

**Tech Stack:** Swift 6, SwiftUI (demo), IOKit/IOHID (accelerometer), Carbon (CGEventTap), AVFoundation (video/audio), QuartzCore (alpha animation), CoreFoundation (run loop).

```swift
    public var keyCombos: [KeyCombo]
    public var volume: Double            // 0.0–1.0

        mode: TriggerMode = .drop,
        keyCombos: [KeyCombo] = [.defaultCombo()],

    ) {
        self.sensitivity = sensitivity
        self.mode = mode
        self.keyCombos = keyCombos
        self.volume = volume
    }

public enum TriggerMode: String, Codable, Sendable {
    case drop
    case hit
}

public struct KeyCombo: Codable, Hashable, Sendable {
    public var modifiers: [CGEventFlags]
    public var keyCode: UInt32

    public init(modifiers: [CGEventFlags], keyCode: UInt32) {
        self.modifiers = modifiers
        self.keyCode = keyCode
    }

    public static func `default`() -> KeyCombo {
        KeyCombo(
            modifiers: [.flagCmd, .flagShift],
            keyCode: UInt32(kVK_T)
        )
    }
}

extension KeyCombo {
    /// Convenience: .cmd(.shift, .t)
    public static func cmd(_ mods: CGEventFlags..., key: CGKeyCode) -> KeyCombo {
        var combo: CGEventFlags = []
        for m in mods { combo.insert(m) }
        return KeyCombo(modifiers: [combo], keyCode: UInt32(key))
    }
}

// MARK: - Persistence

private enum UserDefaultsKeys {
    static let config = "TotemPopSavedConfig"
}

extension TotemConfig {
    /// Load saved config from UserDefaults, merge with provided values.
    /// Explicit init values override saved defaults.
    static func mergedWithSaved() -> TotemConfig {
        let defaults = UserDefaults(suiteName: "com.totempop.settings")
        let data = defaults?.data(forKey: UserDefaultsKeys.config)
        var saved: TotemConfig?
        if let data, let decoded = try? JSONDecoder().decode(TotemConfig.self, from: data) {
            saved = decoded
        }

        // Build merged config: saved values as base, init overrides
        var merged = saved ?? TotemConfig()
        // Note: the caller passes explicit values that will override saved ones
        // This is handled by the caller (TotemPop.init)
        return merged
    }

    func save() {
        let defaults = UserDefaults(suiteName: "com.totempop.settings")
        let data = try? JSONEncoder().encode(self)
        defaults?.set(data, forKey: UserDefaultsKeys.config)
        defaults?.synchronize()
    }
}
```

```swift
import Foundation

/// Token bucket rate limiter: max 5 tokens, refill 1 every 400ms.
/// Each pop() costs 1 token. Prevents spam from sustained motion events.
final actor TotemRateLimiter {
    private let maxTokens: Int = 5
    private let refillInterval: TimeInterval = 0.4  // 400ms
    private var tokens: Int
    private var lastRefill: Date

    init() {
        self.tokens = maxTokens
        self.lastRefill = Date()
    }

    /// Try to consume a token. Returns true if allowed, false if rate-limited.
    func tryConsume() -> Bool {
        refill()
        guard tokens > 0 else { return false }
        tokens -= 1
        return true
    }

    private func refill() {
        let elapsed = Date().timeIntervalSince(lastRefill)
        let newTokens = Int(elapsed / refillInterval)
        if newTokens > 0 {
            tokens = min(maxTokens, tokens + newTokens)
            lastRefill = Date().addingTimeInterval(TimeInterval(newTokens) * refillInterval)
        }
    }
}
```

```swift
import AppKit
import QuartzCore

/// Full-screen click-through overlay window at maximum window level.
/// Clicks pass through. Alpha animation for fade in/out.
final class TotemOverlay {
    private let window: NSWindow
    private let playerLayer: CALayer

    init() {
        // Create a borderless, full-screen window at maximum level
        let style: NSWindow.StyleMask = [.borderless, .fullSizeContentView]
        let backing: NSWindow.BackingStoreType = .buffered
        window = NSWindow(contentRect: NSScreen.main?.frame ?? .zero,
                          styleMask: style,
                          backing: backing,
                          defer: false)

        window.level = .maximum   // kCGMaximumWindowLevel equivalent
        window.backgroundColor = NSColor.clear
        window.ignoresMouseEvents = true
        window.isOpaque = false
        window.hasShadow = false
        window.alphaValue = 0.0

        // Root layer for compositing
        window.contentView?.wantsLayer = true
        window.contentView?.layer = CALayer()
        window.contentView?.layer?.autoresizesSubviews = true
        playerLayer = CALayer()
        playerLayer.frame = (NSScreen.main?.frame ?? .zero)
        playerLayer.autoresizingMask = [.layerWidthSizable, .layerHeightSizable]
        window.contentView?.layer?.addSublayer(playerLayer)
    }

    /// Show the overlay with a fade-in animation.
    func show() {
        guard !window.isVisible else { return }
        window.order(nil)
        window.alphaValue = 0.0
        NSAnimationContext.runAnimationGroup { ctx in
            ctx.duration = 0.15
            window.animator().alphaValue = 1.0
        }
    }

    /// Hide the overlay with a fade-out animation.
    func hide() {
        NSAnimationContext.runAnimationGroup { ctx in
            ctx.duration = 0.2
            window.animator().alphaValue = 0.0
        } completion: {
            window.orderOut(nil)
        }
    }

    /// Set the AVPlayerLayer for video rendering.
    func setPlayerLayer(_ layer: CALayer) {
        playerLayer.sublayers?.forEach { $0.removeFromSuperlayer() }
        layer.frame = (NSScreen.main?.frame ?? .zero)
        playerLayer.addSublayer(layer)
    }
}
```

```swift
import AVFoundation
import QuartzCore

/// Plays the bundled totem video via AVPlayer + AVPlayerLayer.
/// Delegates playback completion to a callback.
final class TotemPlayer {
    private let player: AVPlayer
    private let playerItem: AVPlayerItem
    weak var delegate: TotemPlayerDelegate?

    init(videoResource: String) {
        // Bundle the resource from SPM package resources
        guard let url = Bundle.module.url(forResource: videoResource, withExtension: "mp4") else {
            fatalError("Totem video resource not found: \(videoResource)")
        }
        playerItem = AVPlayerItem(url: url)
        player = AVPlayer(playerItem: playerItem)

        // Observe when playback ends
        NotificationCenter.default.addObserver(
            self,
            selector: #selector(playbackDidFinish),
            name: NSNotification.Name.AVPlayerItemDidPlayToEndTime,
            object: playerItem
        )
    }

    deinit {
        NotificationCenter.default.removeObserver(self)
    }

    /// Play the video. Calls delegate.playerDidFinish() when done.
    func play(volume: Float) {
        player.volume = volume
        player.play()
    }

    @objc private func playbackDidFinish() {
        delegate?.playerDidFinish()
    }

    /// Create the AVPlayerLayer for rendering.
    func playerLayer() -> CALayer {
        let layer = AVPlayerLayer(player: player)
        layer.videoGravity = .resizeAspect
        return layer
    }

    /// Stop playback immediately.
    func stop() {
        player.pause()
    }
}

protocol TotemPlayerDelegate: AnyObject {
    func playerDidFinish()
}
```

Extracted and adapted from the existing `MotionSensorManager.swift`. This is a simpler, event-driven version that doesn't own a run loop — it exposes magnitude events via a callback.

```swift
import Foundation
import IOKit
import IOKit.hid
import CoreFoundation

/// Reads the Bosch BMI286 IMU accelerometer from Apple Silicon Macs via IOHID.
/// Exposes magnitude events via a closure.
/// Requires sudo (IOHIDDeviceOpen needs root).
final class AccelerometerReader: ObservableObject {

    // MARK: - Published state

    @Published var magnitude: Double = 0.0
    @Published private(set) var isConnected = false

    // MARK: - HID constants

    private static let imuReportLen = 22
    private static let imuDataOffset = 6
    private static let reportBufSize = 4096
    private static let pageVendor: Int = 0xFF00
    private static let usageAccel: Int = 3

    // MARK: - Private

    private var sensorThread: Thread?
    private var reportBuffer: [UInt8]?
    private var deviceRef: IOHIDDevice?
    private var runLoopSource: CFRunLoopSource?

    deinit {
        stop()
    }

    // MARK: - Public

    func start() {
        guard sensorThread == nil else { return }
        let thread = Thread { [weak self] in
            self?.sensorLoop()
        }
        thread.name = "TotemPop.Accelerometer"
        thread.qualityOfService = .userInteractive
        sensorThread = thread
        thread.start()
    }

    func stop() {
        sensorThread?.cancel()
        sensorThread = nil
        if let device = deviceRef {
            IOHIDDeviceClose(device, IOOptionBits(kIOHIDOptionsTypeNone))
            deviceRef = nil
        }
        reportBuffer = nil
        runLoopSource = nil
    }

    // MARK: - Sensor thread

    private func sensorLoop() {
        wakeSPUDrivers()

        guard let device = openAccelerometerDevice() else {
            DispatchQueue.main.async { [weak self] in
                self?.isConnected = false
            }
            return
        }

        deviceRef = device

        var buffer = [UInt8](repeating: 0, count: Self.reportBufSize)
        reportBuffer = buffer

        let selfPtr = Unmanaged.passUnretained(self).toOpaque()

        let callback: IOHIDReportCallback = { context, _, _, _, _, report, reportLen in
            guard let context = context else { return }
            let reader = Unmanaged<AccelerometerReader>.fromOpaque(context).takeUnretainedValue()
            reader.handleReport(report, length: reportLen)
        }

        IOHIDDeviceRegisterInputReportCallback(
            device, &buffer, buffer.count, callback, selfPtr
        )

        IOHIDDeviceScheduleWithRunLoop(
            device, CFRunLoopGetCurrent(), kCFRunLoopDefaultMode as CFString
        )

        // Extract the run loop source and store it so we can invalidate it on stop
        runLoopSource = CFRunLoopSourceCreate(nil, 0, nil) // placeholder; real source comes from IOHIDDevice

        DispatchQueue.main.async { [weak self] in
            self?.isConnected = true
        }

        while !(sensorThread?.isCancelled ?? true) {
            CFRunLoopRunInMode(kCFRunLoopDefaultMode, 0.1, true)
        }
    }

    private func wakeSPUDrivers() {
        guard let matching = IOServiceMatching("AppleSPUHIDDriver") else { return }
        var iterator: io_iterator_t = 0
        guard IOServiceGetMatchingServices(kIOMainPortDefault, matching, &iterator) == KERN_SUCCESS else { return }

        var service = IOIteratorNext(iterator)
        while service != 0 {
            var val: Int32 = 1
            let num = CFNumberCreate(kCFAllocatorDefault, .sInt32Type, &val)
            IORegistryEntrySetCFProperty(service, "SensorPropertyReportingState" as CFString, num)
            IORegistryEntrySetCFProperty(service, "SensorPropertyPowerState" as CFString, num)
            var interval: Int32 = 1000
            let intervalNum = CFNumberCreate(kCFAllocatorDefault, .sInt32Type, &interval)
            IORegistryEntrySetCFProperty(service, "ReportInterval" as CFString, intervalNum)
            IOObjectRelease(service)
            service = IOIteratorNext(iterator)
        }
        IOObjectRelease(iterator)
    }

    private func openAccelerometerDevice() -> IOHIDDevice? {
        guard let matching = IOServiceMatching("AppleSPUHIDDevice") else { return nil }
        var iterator: io_iterator_t = 0
        guard IOServiceGetMatchingServices(kIOMainPortDefault, matching, &iterator) == KERN_SUCCESS else { return nil }

        var foundDevice: IOHIDDevice?
        var service = IOIteratorNext(iterator)

        while service != 0 && foundDevice == nil {
            if let pageValue = IORegistryEntryCreateCFProperty(service, "PrimaryUsagePage" as CFString, kCFAllocatorDefault, 0)?.takeRetainedValue() as? Int,
               let usageValue = IORegistryEntryCreateCFProperty(service, "PrimaryUsage" as CFString, kCFAllocatorDefault, 0)?.takeRetainedValue() as? Int,
               pageValue == Self.pageVendor,
               usageValue == Self.usageAccel {

                let device = IOHIDDeviceCreate(kCFAllocatorDefault, service)
                if let device = device,
                   IOHIDDeviceOpen(device, IOOptionBits(kIOHIDOptionsTypeNone)) == KERN_SUCCESS {
                    foundDevice = device
                }
            }
            IOObjectRelease(service)
            service = IOIteratorNext(iterator)
        }

        IOObjectRelease(iterator)
        return foundDevice
    }

    private func handleReport(_ report: UnsafePointer<UInt8>, length: CFIndex) {
        guard length >= Self.imuReportLen else { return }

        let xRaw = report.withMemoryRebound(to: Int32.self, capacity: 6) {
            Int32(littleEndian: $0[Int(Self.imuDataOffset / 4)])
        }
        let yRaw = report.withMemoryRebound(to: Int32.self, capacity: 6) {
            Int32(littleEndian: $0[Int((Self.imuDataOffset + 4) / 4)])
        }
        let zRaw = report.withMemoryRebound(to: Int32.self, capacity: 6) {
            Int32(littleEndian: $0[Int((Self.imuDataOffset + 8) / 4)])
        }

        let x = Double(xRaw)
        let y = Double(yRaw)
        let z = Double(zRaw)
        let mag = sqrt(x * x + y * y + z * z)

        DispatchQueue.main.async { [weak self] in
            self?.magnitude = mag
        }
    }
}
```

```swift
import Foundation
import ApplicationServices

/// Listens for global key combinations using a CGEventTap.
/// Fires a callback when a configured key combo is pressed.
final class HotkeyMonitor {
    private var eventTap: CFMachPort?
    private var runLoopSource: CFRunLoopSource?
    private var activeCombos: [KeyCombo] = []
    weak var delegate: HotkeyMonitorDelegate?

    /// Start listening for the given key combinations.
    func start(with combos: [KeyCombo]) {
        stop()
        activeCombos = combos
        guard !combos.isEmpty else { return }

        eventTap = CGEvent.tapCreate(
            tap: .cgSessionEventTap,
            place: .headInsertEventTap,
            options: .defaultTap,
            eventsOfInterest: CGEventMask(1 << Int(kCGEventKeyDown)) | CGEventMask(1 << Int(kCGEventKeyUp)),
            callback: { [weak self] _, _, event, _ in
                guard let self = self else { return nil }
                self.handleKeyEvent(event)
                return nil
            },
            userInfo: nil
        )

        if let tap = eventTap {
            runLoopSource = CFMachPortCreateRunLoopSource(nil, tap, 0)
            if let source = runLoopSource {
                CFRunLoopAddSource(CFRunLoopGetCurrent(), source, kCFRunLoopDefaultMode)
            }
            CGEvent.tapEnable(tap: tap, enable: true)
        }
    }

    /// Stop listening.
    func stop() {
        if let tap = eventTap {
            CGEvent.tapEnable(tap: tap, enable: false)
            CFMachPortInvalidate(tap)
        }
        eventTap = nil
        runLoopSource = nil
        activeCombos.removeAll()
    }

    private func handleKeyEvent(_ event: CGEvent) {
        let keyCode = UInt32(event.getIntegerValueField(.keyboardEventKeycode))

        // Only process keyUp events to avoid re-triggering on hold
        guard event.type == .keyUp else { return }

        for combo in activeCombos {
            if combo.keyCode == keyCode {
                // Check if all modifiers match
                let currentMods = event.flags
                for modSet in combo.modifiers {
                    if currentMods.contains(modSet) {
                        delegate?.hotkeyTriggered()
                        return
                    }
                }
            }
        }
    }
}

protocol HotkeyMonitorDelegate: AnyObject {
    func hotkeyTriggered()
}
```

```swift
import Foundation

/// Coordinates accelerometer and hotkey monitoring.
/// When either trigger fires, calls the onPop callback (subject to rate limiting).
final actor TotemTrigger {
    private let accelerometer = AccelerometerReader()
    private let hotkeyMonitor = HotkeyMonitor()
    private let rateLimiter = TotemRateLimiter()
    private let config: TotemConfig

    weak var delegate: TotemTriggerDelegate?

    init(config: TotemConfig) {
        self.config = config
        setupObservers()
    }

    func start() {
        accelerometer.start()
        hotkeyMonitor.start(with: config.keyCombos)
    }

    func stop() {
        accelerometer.stop()
        hotkeyMonitor.stop()
    }

    // MARK: - Private

    private func setupObservers() {
        // Accelerometer magnitude events
        accelerometer.publisher(for: \.magnitude)
            .receive(on: DispatchQueue.global(qos: .userInteractive))
            .sink { [weak self] mag in
                self?.checkMagnitude(mag)
            }
            .storeInWeakSelf()

        // Hotkey events
        hotkeyMonitor.delegate = self
    }

    private func checkMagnitude(_ magnitude: Double) {
        let threshold = TotemConfig.threshold(for: config.sensitivity)
        let shouldTrigger: Bool

        switch config.mode {
        case .drop:
            // Drop: detect sharp spike above threshold
            shouldTrigger = magnitude > threshold
        case .hit:
            // Hit: detect sharp impulse above threshold
            shouldTrigger = magnitude > threshold
        }

        if shouldTrigger {
            Task { [weak self] in
                guard let self = self, await self.rateLimiter.tryConsume() else { return }
                await self.delegate?.totemTriggerRequested()
            }
        }
    }
}

// MARK: - HotkeyMonitorDelegate

extension TotemTrigger: HotkeyMonitorDelegate {
    func hotkeyTriggered() {
        Task { [weak self] in
            guard let self = self, await self.rateLimiter.tryConsume() else { return }
```
